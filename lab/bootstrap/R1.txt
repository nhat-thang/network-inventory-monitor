! R1 - Cisco IOSv - minimal bootstrap (management + SSH only)
! Replace <PASSWORD> with the lab password (same value as NET_PASSWORD in .env)
enable
conf t
hostname R1
ip domain-name lab.local
username admin privilege 15 secret <PASSWORD>
interface GigabitEthernet0/0
 ip address 192.168.100.11 255.255.255.0
 no shutdown
exit
crypto key generate rsa modulus 2048
ip ssh version 2
line vty 0 4
 login local
 transport input ssh
end
write memory
